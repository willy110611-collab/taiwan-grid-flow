  <script>
    // 設置 Swagger UI 中的按鈕屬性以符合無障礙需求
    function updateButtons() {
      const swaggerUI = document.getElementById('swagger-ui');
      if (swaggerUI) {
        const buttons = swaggerUI.querySelectorAll('button');
        buttons.forEach(button => {
          if (button.classList.contains('opblock-summary-control')) {
            button.setAttribute('title', '展開/收合');
            button.setAttribute('aria-label', '展開/收合 API 詳細資訊');
          }
          if (button.classList.contains('expand-operation')) {
            button.setAttribute('title', '展開/收合');
            button.setAttribute('aria-label', '展開/收合 API 列表');
          }
        });
      }
    }

    // 先向 Nuxt API 拿設定
    fetch('/nuxtapi/swagger-config')
      .then(res => res.json())
      .then(config => {
        // 初始化 Swagger UI
        const swaggerJsonUrl = `${config.apiBase}/api/v2/openapi.get`;
        SwaggerUIBundle({
          url: swaggerJsonUrl,
          dom_id: '#swagger-ui',
          deepLinking: true,
          presets: [
            SwaggerUIBundle.presets.apis,
            SwaggerUIStandalonePreset
          ],
          layout: "BaseLayout",
          onComplete: () => {
            setTimeout(updateButtons, 100);
          }
        });
      })
      .catch(err => {
        document.body.innerHTML = `<h2 style="color:red;text-align:center;margin-top:50px;">載入設定失敗: ${err}</h2>`;
      });
  </script>
